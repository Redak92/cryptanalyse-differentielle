\begin{frame}{Introduction à la cryptanalyse différentielle}

    \begin{block}{L'équation différentielle}
        Pour une différence d'entrée $\alpha$ et de sortie $\beta$ :
        \begin{equation*}
            F(x) \oplus F(x \oplus \alpha) = \beta
        \end{equation*}
        Une différentielle $\alpha \rightarrow \beta$ est exploitable si sa probabilité d'apparition $p$ est anormalement élevée.
    \end{block} 

    \vspace{0.2cm}
    \textbf{Seuil théorique minimal de sécurité :} $p = 2^{1-n}$
    
    \vspace{0.2cm}
    \begin{alertblock}{Propriété de symétrie}
        L'opération XOR étant commutative, l'évaluation des paires est symétrique :
        \begin{equation*}
            (x) \oplus (x \oplus \alpha) \iff (x \oplus \alpha) \oplus (x)
        \end{equation*} 
        Conséquence stricte : les solutions apparaissent systématiquement par paires.
    \end{alertblock}
    
\end{frame}

\begin{frame}[t]{Algorithme de Recherche Exhaustive } % =============================================
    \begin{block}{Principe}
        Pour chaque $\alpha \in \{0,1\}^n \setminus \{0\}$ :
        \begin{enumerate}
            \item Parcourir tous les $x \in \{0,1\}^n$
            \item Calculer $\beta = F(x) \oplus F(x \oplus \alpha)$
            \item Incrémenter le compteur pour $(\alpha, \beta)$
        \end{enumerate}
    \end{block}

    \footnotesize\[
    \underbrace {
    \bordermatrix{
        & x_0 & x_1 & \cdots & x_{n-1} \cr
        \alpha_0    & F(x_0) \oplus F(x_0 \oplus \alpha_0) & F(x_1) \oplus F(x_1 \oplus \alpha_0) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_0) \cr
        \alpha_1    & F(x_0) \oplus F(x_0 \oplus \alpha_1) & F(x_1) \oplus F(x_1 \oplus \alpha_1) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_1) \cr
        \vdots      & \vdots & \vdots & \ddots & \vdots \cr
        \alpha_{n-1} & F(x_0) \oplus F(x_0 \oplus \alpha_{n-1}) & F(x_1) \oplus F(x_1 \oplus \alpha_{n-1}) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_{n-1}) \cr
    }}_{\text{\small Espace de taille $2^n \cdot 2^n $} }
    \]
    \begin{itemize}
        \item \textbf{Complexité en temps et en mémoire :} $\mathcal{O}(2^{2n})$
    \end{itemize}
\end{frame}

\begin{frame}{L'Algorithme Standard — Principe}  % =============================================
    Transformation de la recherche déterministe en \textbf{échantillonnage statistique} : pour chaque $\alpha$, on teste $k$ paires aléatoires au lieu de $2^n$.

    \begin{block}{Principe}
        Pour chaque $\alpha \in \{0,1\}^n \setminus \{0\}$:
        \begin{enumerate}
            \item Choisir $k$ valeurs aléatoires de $x \in \{0,1\}^n$
            \item Calculer $\beta = F(x) \oplus F(x \oplus \alpha)$ pour chaque $x$
            \item Signaler tout $\beta$ apparaissant au moins 2 fois
        \end{enumerate}
    \end{block}

    \begin{itemize}
        \item \textbf{Complexité en temps et en mémoire :} $\mathcal{O}(2^{n} \cdot p^{-1})$
    \end{itemize}
\end{frame}

\begin{frame}{L'Algorithme Standard — Détection}  
    
    \begin{block}{Modélisation statistique}
        La recherche sur $k$ paires avec une probabilité de succès $p$ suit une loi de Poisson $\mathcal{P}(\lambda)$ de paramètre $\lambda = k \cdot p$.
    \end{block}

    \vspace{0.2cm}
    \textbf{Dimensionnement de l'échantillon ($k = 4/p$)}
    \begin{itemize}
        \item \textbf{Compromis :} Équilibrer fiabilité de détection et temps de calcul.
        \item \textbf{Paramétrage :} Fixer $\lambda = 4$ permet un seuil d'alerte à $X \ge 2$ occurrences.
    \end{itemize}

    \vspace{0.2cm}
    \begin{exampleblock}{Taux de confiance}
        La probabilité de détecter une vraie différentielle devient :
        \begin{equation*}
            P(X \ge 2) = 1 - e^{-4}(1+4) \approx 0{,}908
        \end{equation*}
        La détection est donc assurée avec 90,8\,\% de certitude.
    \end{exampleblock}
    
\end{frame}

\begin{frame}{L'Algorithme Standard — Rejet du bruit}

    \begin{block}{Modélisation du bruit (exemple : Speck 64, $p = 2^{-13}$)}
        Probabilité d'une coïncidence par hasard : $p_{\text{rand}} \approx 2^{-64}$

        Espérance sur $k = 4/p = 2^{15}$ paires :
        $\lambda_{\text{rand}} = k \cdot p_{\text{rand}} = 2^{-49}$

        Probabilité d'observer $\ge 2$ collisions (Poisson) :
        $P(X \ge 2) \approx \frac{\lambda^2}{2} \approx 2^{-99}$
    \end{block}

    $\Rightarrow$ Un faux positif est \textbf{quasi impossible} : le seuil $count \ge 2$ suffit
    à rejeter le bruit.

    \begin{alertblock}{Limites de l'algorithme}
        Reste limité par le facteur $2^n$ $\Rightarrow$ $2^{64}$ itérations pour Speck 64/128
    \end{alertblock}
\end{frame}
