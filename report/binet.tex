\section{ Cryptanalyse différentielle : des fondements à la détection statistique}

\begin{frame}{Introduction à la cryptanalyse différentielle} % 
    \includegraphics[width=12cm]{figures/eprint_screencap.png}
\end{frame}

\begin{frame}{Introduction à la cryptanalyse différentielle} % =============================================
    \small
    \begin{block}{Différence entre deux messages chiffrés}
        Soit $x$ un message clair $\in \{0,1\}^n$, pour une différence d'entrée  $\alpha$ et une différence de sortie $\beta$ on s'intéresse au solutions de l'équation suivante : 
        \begin{equation*}
            F(x) \oplus F(x \oplus \alpha) = \beta
        \end{equation*}
        Si $\beta$ apparaît avec une probabilité $p$ anormalement élevée pour un $\alpha$ donné, la différentielle $\alpha \rightarrow \beta$ est exploitable.    
    \end{block} 

    Meilleur seuil de probabilité d'apparition de différentielle atteignable : $\frac{1}{2^{n + 1}}$
    
    \begin{block}{Les différentielle apparaissent en paires : \\ }
    Pour la paire de messages $(x \oplus \alpha, (x \oplus \alpha) \oplus \alpha)$ :
        \begin{equation*}
            F(x \oplus \alpha) \oplus F((x \oplus \alpha) \oplus \alpha) = F(x \oplus \alpha) \oplus F(x \oplus \alpha \oplus \alpha) = F(x \oplus \alpha) \oplus F(x) = \beta
        \end{equation*}    
    \end{block}
    
\end{frame}

\subsection{Algorithme Naïf}

\begin{frame}[t]{Algorithme de Recherche Exhaustive } % =============================================
    \begin{block}{Principe}
        Pour chaque $\alpha \in \{0,1\}^n \setminus \{0\}$ :
        \begin{enumerate}
            \item Parcourir tous les $x \in \{0,1\}^n$
            \item Calculer $\beta = F(x) \oplus F(x \oplus \alpha)$
            \item Incrémenter le compteur pour $(\alpha, \beta)$
        \end{enumerate}
    \end{block}

    \footnotesize\[
    \underbrace {
    \bordermatrix{
        & x_0 & x_1 & \cdots & x_{n-1} \cr
        \alpha_0    & F(x_0) \oplus F(x_0 \oplus \alpha_0) & F(x_1) \oplus F(x_1 \oplus \alpha_0) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_0) \cr
        \alpha_1    & F(x_0) \oplus F(x_0 \oplus \alpha_1) & F(x_1) \oplus F(x_1 \oplus \alpha_1) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_1) \cr
        \vdots      & \vdots & \vdots & \ddots & \vdots \cr
        \alpha_{n-1} & F(x_0) \oplus F(x_0 \oplus \alpha_{n-1}) & F(x_1) \oplus F(x_1 \oplus \alpha_{n-1}) & \cdots & F(x_{n-1}) \oplus F(x_{n-1} \oplus \alpha_{n-1}) \cr
    }}_{\text{\small Espace de taille $2^n \cdot 2^n $} }
    \]
    \begin{itemize}
        \item \textbf{Complexité en temps et en mémoire :} $\mathcal{O}(2^{2n})$
    \end{itemize}
\end{frame}

\subsection{Algorithme Standard}

\begin{frame}{L'Algorithme Standard — Principe}  % =============================================
    Transformation de la recherche déterministe en \textbf{échantillonnage statistique} : pour chaque $\alpha$, on teste $k$ paires aléatoires au lieu de $2^n$.

    \begin{block}{Principe}
        Pour chaque $\alpha \in \{0,1\}^n \setminus \{0\}$:
        \begin{enumerate}
            \item Choisir $k$ valeurs aléatoires de $x \in \{0,1\}^n$
            \item Calculer $\beta = F(x) \oplus F(x \oplus \alpha)$ pour chaque $x$
            \item Signaler tout $\beta$ apparaissant au moins 2 fois
        \end{enumerate}
    \end{block}

    \begin{itemize}
        \item \textbf{Complexité en temps et en mémoire :} $\mathcal{O}(2^{n} \cdot p^{-1})$
    \end{itemize}
\end{frame}

\begin{frame}[t]{L'Algorithme Standard — Détection}  % =============================================
    
    \begin{block}{Modélisation de la détection}
        Chaque paire testée a une probabilité $p$ de produire le bon $\beta$ : c'est une épreuve de Bernoulli. Comme $p$ est très petit et $k$ grand, le nombre de succès $X$ suit une loi de Poisson $\mathcal{P}(\lambda)$ avec $\lambda = k \times p$.
    \end{block}

    \begin{block}{Choix de $k = 4/p$ ($\lambda = 4$)}
    $\lambda$ trop petit $\Rightarrow$ détection peu fiable. \\
    $\lambda$ trop grand $\Rightarrow$ nombre de paires testées l'est aussi. \\
    Le choix $\lambda = 4$ offre un bon compromis :
    \begin{equation*}
        P(X \ge 2) = 1 - e^{-\lambda}(1+\lambda) \approx 0{,}9085
    \end{equation*}
        Chaque vraie différentielle est détectée avec $\sim 90\,\%$ de confiance.
    \end{block}
    On signale une différentielle dès qu'elle apparaît au moins 2 fois.
    
\end{frame}

\begin{frame}{L'Algorithme Standard — Rejet du bruit}  % =============================================

    \begin{block}{Bruit (permutation aléatoire, $p_{rand} = \frac{1}{2^{n + 1}}$)}
        Pour Speck 64 ($n=64$, $p=2^{-13}$) : $\lambda_{rand} \approx 2^{-49}$ \\[0.5em]
        Par développement de Taylor : $P(X \ge 2) \approx \frac{\lambda^2}{2} \approx 2^{-99}$ \\[0.5em]
        $\Rightarrow$ Collision très improbable pour $count \ge 2$
    \end{block}

    \begin{alertblock}{Limites de l'algorithme}
        Reste limité par le facteur $2^n$ $\Rightarrow$ $2^{64}$ itérations pour Speck 64/128\\ \phantom{}
    \end{alertblock}
\end{frame}
