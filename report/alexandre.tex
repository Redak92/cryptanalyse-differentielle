\section{Efficacité algorithmique : l'approche par collisions}

\begin{frame}{Optimisation du coût temporel : l'approche par collisions}
    
    \begin{block}{Réduction de la complexité ($n=64, p=2^{-13}$)}
        \centering
        \renewcommand{\arraystretch}{1.3}
        \small
        \begin{tabular}{lccc}
            \toprule
            \textbf{Algorithme} & \textbf{Formule} & \textbf{Opérations} & \textbf{Faisabilité} \\
            \midrule
            Naïf Exhaustif & $2^{2n}$ & $2^{128}$ & Impraticable \\
            Naïf Optimisé  & $2^n \cdot p^{-1}$ & $2^{77}$ & Supercalculateurs \\
            \midrule
            \textbf{Par Collisions} & $\mathbf{2^{n/2} \cdot p^{-1}}$ & $\mathbf{2^{45}}$ & \textbf{Station de travail} \\
            \bottomrule
        \end{tabular}
    \end{block}

    \vspace{0.4cm}

    \begin{itemize}
        \item \textbf{Principe de collision :} Trouver deux messages distincts $x$ et $x'$ produisant la même signature de sortie (sur une fonction dérivée $g_\gamma$).
        \item \textbf{Paradoxe des anniversaires :} Réduction du coût temporel.
    \end{itemize}
    
\end{frame}

% --- DIAPO 2 : CONCEPT DE SIGNATURE ---
\begin{frame}{Le concept de signature : la fonction $g_\gamma$}
    \begin{block}{Définition du substitut}
        On choisit une différence arbitraire $\gamma \neq 0$, indépendante de la différence $\alpha$ recherchée.
    \end{block}
    
    \begin{block}{Fonction dérivée $g_\gamma$}
        Pour chaque message $x$, on calcule sa signature de sortie :
        \[ g_{\gamma}(x) = F(x) \oplus F(x \oplus \gamma) \]
    \end{block}

    \begin{itemize}
        \item L'algorithme ne fixe plus $\alpha$ a priori.
        \item On recherche une \textbf{collision globale} : $g_\gamma(x) = g_\gamma(x')$.
        \item La direction $\alpha$ est révélée par la collision : $\alpha = x \oplus x'$.
    \end{itemize}
\end{frame}

        \node[draw, UPSPrune!60, dashed, thick, rounded corners, fit=(x) (xg) (gx), inner sep=0.4cm] {};

        % ÉTAPE 2 : BLOC DROIT (Entrées) + ALPHA
        % ==========================================
        \onslide<2->{
            \node (xa) at (7,0) {$x \oplus \alpha$};
            \node (xag) at (10,0) {$x \oplus \alpha \oplus \gamma$};

            % Relation gamma (grise)
            \draw[<->, thick, gray] (xa) -- node[above] {$\gamma$} (xag);

            % Chiffrements F
            \node[box] (Fxa) at (7,-2) {$F$};
            \node[box] (Fxag) at (10,-2) {$F$};
            \draw[->, thick] (xa) -- (Fxa);
            \draw[->, thick] (xag) -- (Fxag);

            % Relations transversales Alpha (bleues)
            \draw[<->, thick, blue!80!black] (x) to[bend left=20] node[above] {$\alpha$} (xa);
            \draw[<->, thick, blue!80!black] (xg) to[bend right=20] node[below] {$\alpha$} (xag);
        }

        % ==========================================
        % ÉTAPE 3 : Sorties droites + BETA + COLLISION
        % ==========================================
        \onslide<3->{
            % Sorties f
            \node (fxa) at (7,-4) {$F(x \oplus \alpha)$};
            \node (fxag) at (10,-4) {$F(x \oplus \alpha \oplus \gamma)$};
            \draw[->, thick] (Fxa) -- (fxa);
            \draw[->, thick] (Fxag) -- (fxag);

            % XOR et g_gamma
            \node[xor] (xorR) at (8.5,-5.5) {$\oplus$};
            \draw[->, thick] (fxa) -- (xorR);
            \draw[->, thick] (fxag) -- (xorR);

            \node (gxa) at (8.5,-6.5) {$g_{\gamma}(x \oplus \alpha)$};
            \draw[->, thick] (xorR) -- (gxa);

            % Cadre logique droit
            \node[draw, UPSPrune!60, dashed, thick, rounded corners, fit=(xa) (xag) (gxa), inner sep=0.4cm] {};

            % Relations transversales Beta (bleues)
            \draw[<->, thick, blue!80!black] (fx) to[bend left=20] node[above] {$\beta$} (fxa);
            \draw[<->, thick, blue!80!black] (fxg) to[bend right=20] node[below] {$\beta$} (fxag);

            % Collision finale
            \draw[<->, ultra thick, UPSPrune] (gx) -- node[above, font=\bfseries] {Collision} (gxa);
        }

    \end{tikzpicture}

    \vspace{0.2cm}

    \begin{itemize}
        \item<1-> \textbf{Signature :} $g_\gamma(x)$ capture la différence de sortie induite par $\gamma$.
        \item<2-> \textbf{Parallélisme :} Si $(x, x\oplus\alpha)$ est un couple correct ($\beta$), alors $(x\oplus\gamma, x\oplus\alpha\oplus\gamma)$ l'est aussi avec probabilité $p$.
        \item<3-> \textbf{Collision :} Par linéarité du XOR, les deux $\beta$ s'annulent : la collision est détectée.
    \end{itemize}
\end{frame}

% --- DIAPO 4 : PREUVE ALGÉBRIQUE ---
\begin{frame}{Preuve algébrique de la collision}
    \begin{block}{Double différenciation}
        En calculant le XOR des deux signatures, on réorganise les termes :
        \begin{align*}
            g_\gamma(x) \oplus g_\gamma(x \oplus \alpha) &= \big( F(x) \oplus F(x \oplus \gamma) \big) \oplus \big( F(x \oplus \alpha) \oplus F(x \oplus \alpha \oplus \gamma) \big) \\
            &= \big( F(x) \oplus F(x \oplus \alpha) \big) \oplus \big( F(x \oplus \gamma) \oplus F(x \oplus \alpha \oplus \gamma) \big)
        \end{align*}
    \end{block}
    \begin{block}{Annulation de la différence $\beta$}
        Si les deux couples satisfont la différentielle $\alpha \to \beta$ (quadruplet correct) :
        \begin{equation*}
            g_\gamma(x) \oplus g_\gamma(x \oplus \alpha) = \beta \oplus \beta = 0
        \end{equation*}
    \end{block}
    \begin{itemize}
        \item L'égalité $g_\gamma(x) = g_\gamma(x \oplus \alpha)$ est donc garantie.
    \end{itemize}
\end{frame}

% --- DIAPO 5 : EXEMPLE NUMÉRIQUE ---
\begin{frame}{Exemple numérique (Blocs de 4 bits)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Phase de sondage :}
            \begin{itemize}
                \item Fixons $\gamma = 0x1$
                \item $x_1 = \mathbf{0x2} \to g_\gamma(2) = \mathbf{0x5}$
                \item $x_2 = \mathbf{0xA} \to g_\gamma(A) = \mathbf{0x5}$
            \end{itemize}
            \vspace{0.2cm}
            $\Rightarrow$ Une collision est détectée.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Résultat}
                La direction candidate est :
                \[ \alpha = x_1 \oplus x_2 = 2 \oplus A = \mathbf{0x8} \]
            \end{block}
            \begin{itemize}
                \item On a identifié $\alpha=8$ sans tester les 15 autres possibilités.
                \item On vérifie ensuite la probabilité de $0x8 \to \beta$.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
